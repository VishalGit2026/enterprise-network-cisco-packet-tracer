! =========================================================
! R-HQ - HQ ROUTED EDGE ROUTER
! Enterprise Network Infrastructure
! Cisco Packet Tracer
! =========================================================

enable
configure terminal

! ---------------------------------------------------------
! Interface to SW-CORE
! ---------------------------------------------------------

interface GigabitEthernet0/0
 ip address 10.10.254.1 255.255.255.252
 no shutdown

! ---------------------------------------------------------
! Interface to R-ISP
! ---------------------------------------------------------

interface GigabitEthernet0/1
 ip address 10.10.253.1 255.255.255.252
 no shutdown

! ---------------------------------------------------------
! OSPF - Process 10 / Area 0
! ---------------------------------------------------------

router ospf 10
 router-id 1.1.1.1
 network 10.10.253.0 0.0.0.3 area 0
 network 10.10.254.0 0.0.0.3 area 0

! ---------------------------------------------------------
! HQ Default Route to R-ISP
! ---------------------------------------------------------

ip route 0.0.0.0 0.0.0.0 10.10.253.2

end
write memory

! ---------------------------------------------------------
! VERIFICATION COMMANDS
! ---------------------------------------------------------

show ip interface brief
show ip route
show ip ospf neighbor
show ip route 0.0.0.0
ping 10.10.253.2
ping 10.10.254.2
