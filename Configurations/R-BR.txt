! =========================================================
! R-BR - BRANCH ROUTER
! Router-on-a-Stick + DHCP + OSPF
! Enterprise Network Infrastructure
! Cisco Packet Tracer
! =========================================================

enable
configure terminal

! ---------------------------------------------------------
! Branch Router-on-a-Stick
! ---------------------------------------------------------

interface GigabitEthernet0/0
 no ip address

interface GigabitEthernet0/0.10
 encapsulation dot1Q 10
 ip address 10.20.10.1 255.255.255.0

interface GigabitEthernet0/0.40
 encapsulation dot1Q 40
 ip address 10.20.40.1 255.255.255.0

interface GigabitEthernet0/0.60
 encapsulation dot1Q 60
 ip address 10.20.60.1 255.255.255.0

interface GigabitEthernet0/0.80
 encapsulation dot1Q 80
 ip address 10.20.80.1 255.255.255.0

interface GigabitEthernet0/0.90
 encapsulation dot1Q 90
 ip address 10.20.90.1 255.255.255.0

interface GigabitEthernet0/0.99
 encapsulation dot1Q 99
 ip address 10.20.99.1 255.255.255.0

! ---------------------------------------------------------
! WAN Interface to R-ISP
! ---------------------------------------------------------

interface GigabitEthernet0/1
 ip address 10.10.252.1 255.255.255.252
 no shutdown

! ---------------------------------------------------------
! DHCP Excluded Addresses
! ---------------------------------------------------------

ip dhcp excluded-address 10.20.10.1 10.20.10.20
ip dhcp excluded-address 10.20.40.1 10.20.40.20
ip dhcp excluded-address 10.20.60.1 10.20.60.20
ip dhcp excluded-address 10.20.80.1 10.20.80.20
ip dhcp excluded-address 10.20.90.1 10.20.90.20

! ---------------------------------------------------------
! DHCP - BRANCH USERS
! ---------------------------------------------------------

ip dhcp pool BRANCH-USERS
 network 10.20.10.0 255.255.255.0
 default-router 10.20.10.1
 dns-server 8.8.8.8

! ---------------------------------------------------------
! DHCP - BRANCH IT
! ---------------------------------------------------------

ip dhcp pool BRANCH-IT
 network 10.20.40.0 255.255.255.0
 default-router 10.20.40.1
 dns-server 8.8.8.8

! ---------------------------------------------------------
! DHCP - BRANCH GUEST
! ---------------------------------------------------------

ip dhcp pool BRANCH-GUEST
 network 10.20.60.0 255.255.255.0
 default-router 10.20.60.1
 dns-server 8.8.8.8

! ---------------------------------------------------------
! DHCP - BRANCH IOT
! ---------------------------------------------------------

ip dhcp pool BRANCH-IOT
 network 10.20.80.0 255.255.255.0
 default-router 10.20.80.1
 dns-server 8.8.8.8

! ---------------------------------------------------------
! DHCP - BRANCH WIFI
! ---------------------------------------------------------

ip dhcp pool BRANCH-WIFI
 network 10.20.90.0 255.255.255.0
 default-router 10.20.90.1
 dns-server 8.8.8.8

! ---------------------------------------------------------
! OSPF - Process 10 / Area 0
! ---------------------------------------------------------

router ospf 10
 network 10.10.252.0 0.0.0.3 area 0

! ---------------------------------------------------------
! Branch Default Route to R-ISP
! ---------------------------------------------------------

ip route 0.0.0.0 0.0.0.0 10.10.252.2

end
write memory

! ---------------------------------------------------------
! VERIFICATION COMMANDS
! ---------------------------------------------------------

show ip interface brief
show ip dhcp binding
show ip dhcp pool
show ip ospf neighbor
show ip route
show ip route 0.0.0.0
ping 10.10.252.2
